// 改进后的编译函数
func getFieldId(language string, code string) (fileId string, compileError string, err error) {
    // ... 编译逻辑

    var response []ResponseBody
    err = json.Unmarshal(body, &response)
    if err != nil {
        return "", "", fmt.Errorf("解析响应失败: %v", err)
    }

    if len(response) == 0 {
        return "", "", fmt.Errorf("沙箱响应为空")
    }

    // 检查编译状态
    if response[0].Status != "Accepted" {
        // 编译失败，返回错误信息
        stderr := response[0].Files["stderr"]
        return "", stderr, nil
    }

    // 编译成功，返回文件ID
    switch language {
    case "java":
        return response[0].FileIds["a.java"], "", nil
    case "go":
        return response[0].FileIds["main.go"], "", nil
    case "python":
        return response[0].FileIds["main.py"], "", nil
    }

    return "", "", fmt.Errorf("不支持的语言: %s", language)
}
